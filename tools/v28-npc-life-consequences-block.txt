function npcHelpRequestSpec(n,type=null){
 let kind=type;if(!kind){if((n.injury||0)>=2)kind='healing';else if((n.injury||0)>0&&rand()<.55)kind='herbs';else kind='stones'}
 if(kind==='healing')return {type:'healing',label:'求一份回春散疗伤',cost:1,costText:'回春散 1 份'};
 if(kind==='herbs')return {type:'herbs',label:'求些灵草养伤',cost:2,costText:'灵草 2 株'};
 const cost=clamp(3+Math.floor((n.realmIndex||0)/4),3,10);return {type:'stones',label:'借一笔灵石渡过难关',cost,costText:'灵石 '+cost+' 块'}
}
function createNpcHelpRequest(n,type=null,forced=false){
 if(typeof n!=='object')n=state.npcs.find(x=>x.id===Number(n));if(!n||!n.alive||state.flags.dead)return null;ensureNpcConsequenceShape();const w=state.world,today=dayNumber();if(w.npcPendingRequest&&!forced)return null;if(!forced){if(!n.known||n.location!==state.player.location||(n.relation||0)<25)return null;if(n.lastRequestDay&&today-n.lastRequestDay<60)return null}
 const spec=npcHelpRequestSpec(n,type);n.lastRequestDay=today;w.npcPendingRequest={id:'npc-request-'+today+'-'+n.id,npcId:n.id,type:spec.type,label:spec.label,cost:spec.cost,costText:spec.costText,createdDay:today,expiresDay:today+7};w.npcConsequenceEvents=(w.npcConsequenceEvents||0)+1;n.known=true;addPersonal('【故人求援】'+n.name+'找到你，希望你能'+spec.label+'。','major');return {...w.npcPendingRequest}
}
function expireNpcHelpRequest(){
 ensureNpcConsequenceShape();const req=state.world.npcPendingRequest;if(!req||dayNumber()<=req.expiresDay)return false;const n=state.npcs.find(x=>x.id===req.npcId);state.world.npcPendingRequest=null;state.player.npcRequestsRefused=(state.player.npcRequestsRefused||0)+1;if(n&&n.alive){n.relation=clamp((n.relation||0)-4,-100,100);if((n.injury||0)>=3&&rand()<.15)npcDie(n,'重伤求援未果后伤势恶化');else if(n.known)addPersonal('【求援失约】你没有在期限内回应'+n.name+'，这段交情冷了几分。','bad')}return true
}
function resolveNpcHelpRequest(choice='help'){
 ensureNpcConsequenceShape();const p=state.player,w=state.world,req=w.npcPendingRequest;if(!req)return 'none';const n=state.npcs.find(x=>x.id===req.npcId);if(!n||!n.alive){w.npcPendingRequest=null;save();render();return 'gone'}if(dayNumber()>req.expiresDay){expireNpcHelpRequest();save();render();return 'expired'}
 if(choice!=='help'){w.npcPendingRequest=null;p.npcRequestsRefused=(p.npcRequestsRefused||0)+1;n.relation=clamp((n.relation||0)-6,-100,100);if((n.injury||0)>=3&&rand()<.20)npcDie(n,'重伤求援被拒后伤势恶化');else addPersonal('你拒绝了'+n.name+'的求援。对方没有再多说，但关系明显疏远。','bad');save();render();showResult('已经拒绝',n.name+' 关系 -6','bad');return 'refused'}
 if(req.type==='healing'){if((p.healingPills||0)<req.cost){showResult('回春散不足','需要 '+req.cost+' 份回春散。','bad');return 'insufficient'}p.healingPills-=req.cost;n.injury=Math.max(0,(n.injury||0)-2)}
 else if(req.type==='herbs'){if((p.herbs||0)<req.cost){showResult('灵草不足','需要 '+req.cost+' 株灵草。','bad');return 'insufficient'}p.herbs-=req.cost;n.injury=Math.max(0,(n.injury||0)-1);n.wealth=(n.wealth||0)+1}
 else{if((p.spiritStones||0)<req.cost){showResult('灵石不足','需要 '+req.cost+' 块灵石。','bad');return 'insufficient'}p.spiritStones-=req.cost;n.wealth=(n.wealth||0)+req.cost}
 const relGain=req.type==='healing'?12:9,favorGain=req.type==='healing'?2:1;n.relation=clamp((n.relation||0)+relGain,-100,100);n.favorDebt=(n.favorDebt||0)+favorGain;p.npcRequestsHelped=(p.npcRequestsHelped||0)+1;w.npcPendingRequest=null;addPersonal('你帮了'+n.name+'一把。对方把这份人情记在心里。','good');save();render();showResult('援手已结算',req.costText+'\n关系 +'+relGain+'\n人情债 +'+favorGain,'good');return 'helped'
}

function simulateNpcFeud(a,b,outcome=null,forced=false){
 if(typeof a!=='object')a=state.npcs.find(x=>x.id===Number(a));if(typeof b!=='object')b=state.npcs.find(x=>x.id===Number(b));if(!a||!b||a.id===b.id||!a.alive||!b.alive)return null;ensureNpcConsequenceShape();const today=dayNumber(),score=npcRelationBetween(a,b);if(!forced&&score>-40)return null;if(!forced&&((a.lastConflictDay&&today-a.lastConflictDay<90)||(b.lastConflictDay&&today-b.lastConflictDay<90)))return null;

 else{loser.injury=clamp((loser.injury||0)+1,0,3);const spoils=Math.min(loser.wealth||0,rint(1,4));loser.wealth=Math.max(0,(loser.wealth||0)-spoils);winner.wealth=(winner.wealth||0)+spoils;changeNpcRelationPair(a,b,-8,'冲突升级');if(a.known||b.known)addWorld('【仇怨升级】'+winner.name+'与'+loser.name+'在'+a.location+'大打出手，'+loser.name+'负伤退走。','bad');result='injury'}

}
