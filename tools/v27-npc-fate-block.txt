function ensureNpcLifeShape(){
 const p=state.player,w=state.world;if(p.npcBattleAssists==null)p.npcBattleAssists=0;if(p.npcFriendDeaths==null)p.npcFriendDeaths=0;if(!w.npcRelations||typeof w.npcRelations!=='object'||Array.isArray(w.npcRelations))w.npcRelations={};if(!Array.isArray(w.npcMemorials))w.npcMemorials=[];if(w.npcLifeEvents==null)w.npcLifeEvents=0;
 for(const n of state.npcs){if(n.bornDay==null)n.bornDay=1-(Number(n.age)||20)*360;if(n.injury==null)n.injury=0;if(n.adventures==null)n.adventures=0;if(n.secretRealmTrips==null)n.secretRealmTrips=0;if(n.factionBattles==null)n.factionBattles=0;if(n.deathCause===undefined)n.deathCause=null;if(n.deathDay==null)n.deathDay=0;if(n.lastFateDay==null)n.lastFateDay=0;if(n.lastSecretRealmKey===undefined)n.lastSecretRealmKey=null;if(n.lastFactionLifeDay==null)n.lastFactionLifeDay=0;if(n.battleAssists==null)n.battleAssists=0;n.age=npcLifeAge(n)}
}
function npcLifeAge(n){return Math.max(0,Math.floor((dayNumber()-(n?.bornDay||1))/360))}
function npcInjuryText(n){const i=clamp(Number(n?.injury)||0,0,3);return ['无伤','轻伤','重伤','濒危'][i]}
function npcPairKey(a,b){const x=Number(a?.id||a),y=Number(b?.id||b);return x<y?x+'-'+y:y+'-'+x}
function npcRelationBetween(a,b){ensureNpcLifeShape();return clamp(Number(state.world.npcRelations[npcPairKey(a,b)])||0,-100,100)}
function changeNpcRelationPair(a,b,delta,reason=''){ensureNpcLifeShape();if(!a||!b||a.id===b.id)return 0;const key=npcPairKey(a,b),old=npcRelationBetween(a,b),next=clamp(old+Math.trunc(delta||0),-100,100);state.world.npcRelations[key]=next;if((old<40&&next>=40)||(old>-40&&next<=-40)){const bond=next>=40?'结为可靠同道':'结下明显仇怨';if(a.known||b.known)addWorld('【人物关系】'+a.name+'与'+b.name+'经过多次往来，'+bond+(reason?'（'+reason+'）':'')+'。',next>=40?'good':'bad')}return next}
function npcMemorial(n,cause){ensureNpcLifeShape();const entry={id:n.id,name:n.name,faction:n.faction,realmIndex:n.realmIndex,age:npcLifeAge(n),cause:cause||'身死道消',day:dayNumber(),relation:n.relation||0,grudge:n.grudge||0};state.world.npcMemorials.unshift(entry);state.world.npcMemorials=state.world.npcMemorials.slice(0,60);return entry}
function npcDie(n,cause='身死道消'){if(!n||!n.alive)return false;ensureNpcLifeShape();n.alive=false;n.deathCause=cause;n.deathDay=dayNumber();n.injury=3;n.age=npcLifeAge(n);npcMemorial(n,cause);state.world.npcLifeEvents=(state.world.npcLifeEvents||0)+1;if(n.known){addWorld('【人物命运】'+n.name+'（'+n.faction+'）'+cause+'，终年 '+n.age+' 岁。','bad');if((n.relation||0)>=40){state.player.npcFriendDeaths=(state.player.npcFriendDeaths||0)+1;addPersonal('【故人陨落】与你交情深厚的'+n.name+'已经'+cause+'。这段交情只剩旧日记录。','bad')}else if((n.grudge||0)>=45)addPersonal('【旧怨终结】与你积怨已深的'+n.name+'已经'+cause+'，这段恩怨随之终止。','major')}return true}
function npcAdventureDifficulty(n){const loc=LOCATIONS[n.location],danger=loc?.danger||.2;return danger*10+Math.max(0,majorRealmStage(n.realmIndex)-1)}
function simulateNpcAdventure(n,outcome=null){if(typeof n!=='object')n=state.npcs.find(x=>x.id===Number(n));if(!n||!n.alive)return null;ensureNpcLifeShape();n.adventures=(n.adventures||0)+1;n.lastFateDay=dayNumber();state.world.npcLifeEvents=(state.world.npcLifeEvents||0)+1;const danger=npcAdventureDifficulty(n),successChance=clamp(.78+n.realmIndex*.018+n.courage/700-danger*.055-(n.injury||0)*.13,.16,.94);let result=outcome;if(result==null)result=rand()<successChance?'success':((n.injury||0)>=2||danger>=6&&rand()<.16?'death':'injury');if(result==='success'){const gain=rint(3,8)+Math.floor(n.ambition/25);n.progress+=gain;n.wealth=(n.wealth||0)+rint(1,4);if(n.known&&rand()<.28)addWorld('【人物历练】'+n.name+'在'+n.location+'历练有所得，修为与身家都有增长。');return 'success'}if(result==='death'){npcDie(n,'外出历练时陨落');return 'death'}n.injury=clamp((n.injury||0)+1,0,3);if(n.known&&rand()<.38)addWorld('【人物历练】'+n.name+'在'+n.location+'遭遇凶险，带伤而归。','bad');return 'injury'}
function npcSecretRealmKey(r){return r?(r.id||r.name||'secret')+'@'+(r.openDay||r.startDay||state.world.secretRealmCount||0):null}
function simulateNpcSecretRealm(n,outcome=null){if(typeof n!=='object')n=state.npcs.find(x=>x.id===Number(n));const r=currentSecretRealm();if(!n||!n.alive||!r||n.location!==r.location)return null;ensureNpcLifeShape();const key=npcSecretRealmKey(r);if(n.lastSecretRealmKey===key)return 'already';n.lastSecretRealmKey=key;n.secretRealmTrips=(n.secretRealmTrips||0)+1;state.world.npcLifeEvents=(state.world.npcLifeEvents||0)+1;const threat=Number(r.threat)||0,successChance=clamp(.70+(n.realmIndex-threat)*.08+n.courage/900-(n.injury||0)*.12,.10,.94);let result=outcome;if(result==null)result=rand()<successChance?'success':((threat-n.realmIndex>=3||n.injury>=2)&&rand()<.28?'death':'injury');if(result==='success'){n.progress+=rint(6,14);n.wealth=(n.wealth||0)+rint(2,7);if(n.known)addWorld('【秘境命运】'+n.name+'从'+r.name+'活着归来，得了一场机缘。','good');return 'success'}if(result==='death'){npcDie(n,'探索'+r.name+'时陨落');return 'death'}n.injury=clamp((n.injury||0)+1,0,3);if(n.known)addWorld('【秘境命运】'+n.name+'从'+r.name+'负伤逃出。','bad');return 'injury'}
function simulateNpcFactionBattle(n,outcome=null){if(typeof n!=='object')n=state.npcs.find(x=>x.id===Number(n));const m=factionWar();if(!n||!n.alive||!m||m.status!=='active'||!['青云宗','血刀门'].includes(n.faction))return null;ensureNpcLifeShape();if(n.lastFactionLifeDay===dayNumber()&&outcome==null)return 'already';n.lastFactionLifeDay=dayNumber();n.factionBattles=(n.factionBattles||0)+1;state.world.npcLifeEvents=(state.world.npcLifeEvents||0)+1;if(n.location!==m.location)n.location=m.location;const sidePower=n.faction==='青云宗'?state.world.qingyunPower:state.world.bloodBladePower,enemyPower=n.faction==='青云宗'?state.world.bloodBladePower:state.world.qingyunPower,chance=clamp(.56+(sidePower-enemyPower)/220+n.realmIndex*.012+n.courage/1000-(n.injury||0)*.12,.18,.88);let result=outcome;if(result==null)result=rand()<chance?'success':((n.injury||0)>=2||rand()<.16?'death':'injury');if(result==='success'){n.progress+=rint(3,9);n.wealth=(n.wealth||0)+rint(1,4);if(n.known&&rand()<.32)addWorld('【宗门战场】'+n.name+'替'+n.faction+'出战后全身而退。');return 'success'}if(result==='death'){npcDie(n,'在青云宗与血刀门的大战中战死');return 'death'}n.injury=clamp((n.injury||0)+1,0,3);if(n.known&&rand()<.45)addWorld('【宗门战场】'+n.name+'在临江城大战中负伤退下。','bad');return 'injury'}
function simulateNpcNetworkTick(force=false){ensureNpcLifeShape();if(!force&&dayNumber()%15!==0)return 0;let count=0;const byLoc={};for(const n of state.npcs){if(!n.alive)continue;(byLoc[n.location]||(byLoc[n.location]=[])).push(n)}for(const arr of Object.values(byLoc)){if(arr.length<2)continue;if(!force&&rand()>.42)continue;const a=pick(arr),others=arr.filter(x=>x.id!==a.id),b=pick(others);if(!b)continue;let delta;if(a.faction===b.faction)delta=rint(1,5);else if(new Set([a.faction,b.faction]).has('青云宗')&&new Set([a.faction,b.faction]).has('血刀门'))delta=-rint(3,8);else delta=rand()<.62?rint(1,4):-rint(1,4);if((a.ambition+b.ambition)>160&&rand()<.35)delta-=rint(1,4);changeNpcRelationPair(a,b,delta,a.location+'相遇');count++}return count}
function simulateNPCsV27(){ensureNpcLifeShape();const today=dayNumber(),r=currentSecretRealm(),war=factionWar();for(const n of state.npcs){if(!n.alive)continue;n.age=npcLifeAge(n);if(n.age>=realmLifespanFloor(n.realmIndex)){npcDie(n,'寿元耗尽坐化');continue}if(n.injury>0&&today%20===0&&rand()<.48)n.injury=Math.max(0,n.injury-1);const injuryM=[1,.82,.56,.30][clamp(n.injury||0,0,3)];n.progress+=(n.talent/100)*npcCultivationRate(n.realmIndex)*(.70+rand()*.60)*injuryM;if(n.progress>=100&&n.realmIndex<REALMS.length-1){n.progress-=100;n.realmIndex++;if(rand()<.12||n.known){n.known=true;addWorld(n.name+'突破至'+REALMS[n.realmIndex].name+'。')}}if(rand()<.035)n.location=npcMoveTarget(n);if(today%15===0&&today-(n.lastFateDay||0)>=15&&rand()<clamp(.035+n.courage/1800+(LOCATIONS[n.location]?.danger||0)*.08,.04,.16))simulateNpcAdventure(n);if(n.alive&&r&&n.location===r.location&&n.lastSecretRealmKey!==npcSecretRealmKey(r)&&rand()<clamp(.015+n.ambition/2200+n.courage/3000,.02,.10))simulateNpcSecretRealm(n);if(n.alive&&war&&war.status==='active'&&['青云宗','血刀门'].includes(n.faction)&&today-(n.lastFactionLifeDay||0)>=1&&rand()<clamp(.025+n.courage/2200,.03,.09))simulateNpcFactionBattle(n)}simulateNpcNetworkTick()}
function findNpcBattleAlly(enemy){ensureNpcLifeShape();if(!state||state.flags.dead||enemy?.nonLethal)return null;const forced=Number(state.flags.forceNpcAllyId)||0;if(forced){state.flags.forceNpcAllyId=null;const n=state.npcs.find(x=>x.id===forced);if(n&&n.alive&&n.location===state.player.location&&n.injury<3)return n}const candidates=state.npcs.filter(n=>n.alive&&n.known&&n.location===state.player.location&&(n.relation||0)>=50&&(n.grudge||0)<20&&(n.injury||0)<3);if(!candidates.length)return null;candidates.sort((a,b)=>(b.relation+a.realmIndex*3)-(a.relation+b.realmIndex*3));const best=candidates[0],chance=clamp(.18+(best.relation||0)/180+best.courage/700,.35,.82);return rand()<chance?best:null}
function npcBattleAllyTurn(enemy){if(!combat||!combat.allyNpcId||combat.allyTurns<=0)return;const n=state.npcs.find(x=>x.id===combat.allyNpcId);if(!n||!n.alive){combat.allyNpcId=null;combat.allyTurns=0;return}const base=7+n.realmIndex*4+Math.floor(n.talent/12)+rint(3,9),dmg=Math.max(1,Math.floor(base*realmSuppressionMultiplier(n.realmIndex,enemy.realm||0)));combat.enemyHp-=dmg;combat.allyTurns--;if(!combat.allyCounted){combat.allyCounted=true;state.player.npcBattleAssists=(state.player.npcBattleAssists||0)+1;n.battleAssists=(n.battleAssists||0)+1;addPersonal('【故人助阵】'+n.name+'在你交战时出手相助。','good')}combat.logs.push(n.name+'从旁助阵，造成 '+dmg+' 点伤害。')}
function npcNetworkHighlights(){ensureNpcLifeShape();const out=[];for(const [key,scoreRaw] of Object.entries(state.world.npcRelations)){const score=Number(scoreRaw)||0;if(Math.abs(score)<25)continue;const [aId,bId]=key.split('-').map(Number),a=state.npcs.find(x=>x.id===aId),b=state.npcs.find(x=>x.id===bId);if(!a||!b||(!a.known&&!b.known))continue;out.push({a:a.name,b:b.name,score})}return out.sort((x,y)=>Math.abs(y.score)-Math.abs(x.score)).slice(0,8)}
