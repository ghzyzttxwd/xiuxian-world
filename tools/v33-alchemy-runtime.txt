 return syncV33AlchemyState()

function v33AlchemyChance(r){const p=state.player,prof=Number(p.alchemyProf)||0,realmEdge=Math.max(0,r.unlock-p.realmIndex)*.02,sect=p.sectRank==='内门弟子'?.04:p.sectRank==='真传弟子'?.06:0;return clamp((r.base||.5)+Math.min(.34,prof/900)+sect-realmEdge,.22,.96)}
function v33RollGrade(r,forced=null){if(forced&&V33_GRADE_META[forced])return forced;const prof=Number(state.player.alchemyProf)||0,x=rand(),high=r.unlock>=15;if(x<clamp((prof-(high?260:120))/2600,.01,.10))return 'perfect';if(x<clamp(.05+prof/1500,.06,.24))return 'superior';if(x<clamp(.22+prof/1000,.24,.54))return 'fine';return 'common'}
function brewV33Alchemy(id,opts={}){ensureV33AlchemyShape();const r=V33_RECIPE_BY_ID[id],p=state.player;if(!r)return 'missing';if(!p.knownRecipeIds.includes(id)&&!opts.force)return 'unknown';if(!opts.force&&p.realmIndex<r.unlock)return 'realm';if(!opts.force&&!v33HasIngredients(r))return 'cost';if(!opts.force)v33SpendIngredients(r);p.alchemyBatches=(p.alchemyBatches||0)+1;if(!opts.noTime){advanceDays(r.days||1);if(state.flags.dead)return 'dead'}const chance=opts.forceSuccess===true?1:opts.forceSuccess===false?0:v33AlchemyChance(r),success=rand()<chance;if(!success){p.alchemyProf=(p.alchemyProf||0)+(r.unlock>=15?6:4);p.alchemyFailures++;addPersonal('炼制【'+r.name+'】失败，药性冲突，材料尽毁。','bad');save();render();return 'failed'}const grade=v33RollGrade(r,opts.grade),stack=p.pillInventoryById[id];stack[grade]=(stack[grade]||0)+1;p.alchemyProf=(p.alchemyProf||0)+(r.unlock>=19?16:r.unlock>=15?13:8);p.alchemySuccesses++;const old=p.alchemyBestQualityByRecipe[id],oldRank=old?(V33_GRADE_META[old]?.rank??-1):-1;if(V33_GRADE_META[grade].rank>oldRank)p.alchemyBestQualityByRecipe[id]=grade;addPersonal('【炼丹】'+r.name+'炼成，品质【'+V33_GRADE_META[grade].name+'】。','good');save();render();return grade}

function useV33Pill(id,grade=null,force=false){ensureV33AlchemyShape();const r=V33_RECIPE_BY_ID[id],p=state.player;if(!r)return 'missing';grade=grade||v33HighestPillGrade(id);if(!force&&(!grade||(p.pillInventoryById[id][grade]||0)<=0))return 'none';if(!grade)grade='common';if(!force&&p.pillToxicity>=100&&!(r.effect?.toxicity<0))return 'toxicity';if(r.effect?.path&&p.daoPath!==r.effect.path&&!force)return 'path';if(r.effect?.maxUses&&p.lifeExtensionPillsUsed>=r.effect.maxUses&&!force)return 'maxUses';if(!force)p.pillInventoryById[id][grade]--;const meta=V33_GRADE_META[grade]||V33_GRADE_META.common,tolerance=Number(p.pillToleranceById[id])||0,toxAdd=Math.max(0,Math.round((r.toxicity||0)*meta.tox+tolerance*1.5)),toxPenalty=p.pillToxicity>=80?.66:p.pillToxicity>=55?.82:1,scale=meta.power*toxPenalty,e=r.effect||{};let notes=[];
 if(e.hp){const before=p.hp,gain=Math.max(1,Math.floor(maxHp()*e.hp*scale));p.hp=Math.min(maxHp(),p.hp+gain);notes.push('气血 +'+(p.hp-before))}
 if(e.qi){const before=p.qi,gain=Math.max(1,Math.floor(maxQi()*e.qi*scale));p.qi=Math.min(maxQi(),p.qi+gain);notes.push('灵力 +'+(p.qi-before))}
 if(e.injury){const before=p.injury||0;p.injury=clamp(before+Math.round(e.injury),0,3);if(before!==p.injury)notes.push('伤势 '+before+'→'+p.injury)}
 if(e.progress){const before=p.progress,gain=Math.max(1,Math.floor(realm().need*e.progress*scale));p.progress=Math.min(realm().need,p.progress+gain);notes.push('修为 +'+(p.progress-before))}
 if(e.manualProf){p.manualProf+=(e.manualProf*scale)|0;notes.push('功法熟练 +'+((e.manualProf*scale)|0))}
 if(e.insight){const n=Math.max(1,Math.floor(e.insight*scale));p.insight+=n;notes.push('悟道 +'+n)}
 if(e.toxicity){p.pillToxicity=clamp(p.pillToxicity+Math.round(e.toxicity*scale),0,120);notes.push('丹毒 '+(e.toxicity<0?'下降':'变化'))}
